function! g:TermdebugToggleBp()
    local file = expand('%:p')
    local line_nr = line('.')
    local bp_out = TermDebugSendCommand('info breakpoints')
    local pattern = '\<'.file.'\:' . line_nr . '\>'
    if bp_out =~ pattern
        let match_line = matchstr(bp_out, '^\s*\d\+\ze.*' . pattern)
        let bp_num = str2nr(match_line)
        call TermDebugSendCommand('delete ' . bp_num)
        echo "Breakpoint removed: line ".line_nr
    else
        call TermDebugSendCommand('break ' . file . ':' . line_nr)
        echo "Breakpoint set: line ".line_nr
    endif
endfunction

" -----------------------------------------------------------------------------
" Gemini C‑code assistant
" \gn       reset chat session
" \ga       visual‑select: audit snippet
" \gA       normal mode: audit whole file
" \gr       visual‑select: refactor selected code
" \gg       normal mode: manual prompt input, insert generated code below cursor
" \gc       normal mode: multi‑line // comment block as requirement, insert code after block
" -----------------------------------------------------------------------------
nnoremap <leader>gn :!./gemini.sh new<CR>

vnoremap <leader>ga :<C-u>call g:GeminiAuditSel()<CR>
function! g:GeminiAuditSel() range
    let prompt = 'Audit this snippet, find undefined‑behavior, memory‑safety defects, type risks. Output concise conclusions.'
    silent "'<,'>w !./gemini.sh ask -"
endfunction

nnoremap <leader>gA :call g:GeminiAuditFile()<CR>
function! g:GeminiAuditFile()
    let prompt = 'Audit this complete source file, find undefined‑behavior, memory leaks, buffer‑overflow, type‑safety issues. Give short conclusion list.'
    echo system("printf '%s\n' " . shellescape(prompt) . " | cat - " . shellescape(expand('%')) . " | ./gemini.sh ask -")
endfunction

vnoremap <leader>gr :<C-u>call g:GeminiRefactorSel()<CR>
function! g:GeminiRefactorSel() range
    let prompt = 'Refactor this code, preserve original logic, improve safety and readability.'
    silent "'<,'>w !./gemini.sh ask -"
endfunction

nnoremap <leader>gg :call g:GeminiGenQuery()<CR>
function! g:GeminiGenQuery()
    let user_input = input("Generate code prompt: ")
    if empty(user_input)
        return
    endif
    let prompt = 'Write concise standard code for: ' . user_input
    let resp = system("printf '%s\n' " . shellescape(prompt) . " | ./gemini.sh ask -")
    call append(line('.'), split(resp, "\n"))
endfunction

function! g:GetMultiLineComment()
    let start = line('.')
    let endln = line('.')
    let l = start
    while l >= 1
        let ln = getline(l)
        if ln =~ '^\s*//'
            let start = l
            let l -= 1
        else
            break
        endif
    endwhile
    let l = endln + 1
    while l <= line('$')
        let ln = getline(l)
        if ln =~ '^\s*//'
            let endln = l
            let l += 1
        else
            break
        endif
    endwhile
    let block_text = join(getline(start, endln), "\n")
    return {'text': block_text, 'end_line': endln}
endfunction

nnoremap <leader>gc :call g:GeminiGenFromComment()<CR>
function! g:GeminiGenFromComment()
    let comm = g:GetMultiLineComment()
    let comment_block = comm.text
    let block_end = comm.end_line
    if empty(comment_block)
        echo "Error: could not find // comment block under cursor"
        return
    endif
    let prompt = 'Following this multi‑line requirement comment, write compact standard implementation. Requirement:' . comment_block
    let resp = system("printf '%s\n' " . shellescape(prompt) . " | ./gemini.sh ask -")
    call append(block_end, split(resp, "\n"))
    echo "Code inserted after multi‑line comment block"
endfunction
